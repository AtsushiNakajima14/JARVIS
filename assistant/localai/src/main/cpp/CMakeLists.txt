cmake_minimum_required(VERSION 3.22.1)
project(jarvis_local_llm LANGUAGES C CXX)
# Personal APKs are debug-signed; inference still needs optimized native kernels.
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)

get_filename_component(JARVIS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(LLAMA_SOURCE "${JARVIS_ROOT}/.tooling/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(jarvis_llama_source
        URL https://github.com/ggml-org/llama.cpp/archive/bb4caa7540188872173c44d161602d9271386413.zip
        URL_HASH SHA256=5c588d37f6d6c6d25b142b5003cff2ab6b94ed676cdf56638d56836182c909aa
    )
    FetchContent_GetProperties(jarvis_llama_source)
    if(NOT jarvis_llama_source_POPULATED)
        FetchContent_Populate(jarvis_llama_source)
    endif()
    set(LLAMA_SOURCE "${jarvis_llama_source_SOURCE_DIR}")
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS localai_compile=8 localai_link=1)
set(CMAKE_JOB_POOL_COMPILE localai_compile)
set(CMAKE_JOB_POOL_LINK localai_link)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)

add_library(jarvis_local_llm SHARED local_llm.cpp)
target_compile_features(jarvis_local_llm PRIVATE cxx_std_17)
target_compile_options(jarvis_local_llm PRIVATE -O2)
target_link_options(jarvis_local_llm PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(jarvis_local_llm PRIVATE llama android log)
